尾部的零

求一个数尾部的0的个数

long count(long n)
{
    if (n < 0)
    {
        return -1;
    }
    long num = 0;
    long tail = 1;
    while (true)
    {
        tail = tail * 10;
        if (n % tail == 0)
        {
            num += 1;
        }else
        {
            break;
        }
    }
    return num;
}

求n的阶乘的尾部0的个数

long long count(long long n)
{
    if (n < 0)
    {
        return -1;
    }
    long long num = 0;
    while ((n/=5) > 0)
    {
        num += n;
    }
    return num;
}
